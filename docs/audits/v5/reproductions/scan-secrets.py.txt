"""Read-only pattern scan of current files, reachable Git blobs, and build; redact values."""
import subprocess,re,json,sys,base64
from pathlib import Path
repo=Path(sys.argv[1]);dest=Path(sys.argv[2])
patterns={
 'github_token':rb'\b(?:gh[pousr]_[A-Za-z0-9]{30,255}|github_pat_[A-Za-z0-9_]{50,255})\b',
 'supabase_management_token':rb'\bsbp_[a-f0-9]{30,80}\b',
 'supabase_secret_key':rb'\bsb_secret_[A-Za-z0-9_-]{20,200}\b',
 'private_key':rb'-----BEGIN (?:RSA |EC |OPENSSH |DSA )?PRIVATE KEY-----',
 'aws_access_key_id':rb'\b(?:AKIA|ASIA)[A-Z0-9]{16}\b',
 'postgres_password_url':rb'postgres(?:ql)?://[^\s:/]+:([^\s@]{8,})@[^\s/]+',
 'jwt':rb'\beyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{20,}\b'
}
patterns={k:re.compile(v) for k,v in patterns.items()};found=[];public=0
def scan(data,path,scope,oid=None):
 global public
 if b'\0' in data[:8192]:return
 for kind,rx in patterns.items():
  for m in rx.finditer(data):
   category=kind
   if kind=='jwt':
    try:
     p=m[0].split(b'.')[1];claims=json.loads(base64.urlsafe_b64decode(p+b'='*((-len(p))%4)))
     if claims.get('role')=='anon':public+=1;continue
     if claims.get('role')=='service_role':category='service_role_jwt'
    except Exception:pass
   if kind=='postgres_password_url' and any(x in m[0].lower() for x in [b'password',b'example',b'localhost',b'127.0.0.1',b'votre',b'***']):continue
   found.append({'type':category,'scope':scope,'path':path,'line':data[:m.start()].count(b'\n')+1,'object':oid,'value':'[REDACTED]'})
files=subprocess.check_output(['git','ls-files','-z'],cwd=repo).split(b'\0'); current=0
for f in files:
 if not f:continue
 p=f.decode();scan((repo/p).read_bytes(),p,'current');current+=1
raw=subprocess.check_output(['git','rev-list','--objects','--all'],cwd=repo).decode().splitlines()
proc=subprocess.Popen(['git','cat-file','--batch'],cwd=repo,stdin=subprocess.PIPE,stdout=subprocess.PIPE)
blobs=0;totalbytes=0
for entry in raw:
 oid,_,path=entry.partition(' ')
 proc.stdin.write((oid+'\n').encode());proc.stdin.flush()
 h=proc.stdout.readline().decode().split();size=int(h[2]);data=proc.stdout.read(size);proc.stdout.read(1)
 if h[1]=='blob':blobs+=1;totalbytes+=size;scan(data,path,'history',oid)
proc.stdin.close();proc.wait()
built=0
for p in (Path(sys.argv[3]) if len(sys.argv)>3 else repo/'dist').rglob('*'):
 if p.is_file():scan(p.read_bytes(),p.name,'build');built+=1
result={'method':'known-format pattern matching, no secret values output; no entropy or validity verification','tracked_files':current,'unique_historical_blobs':blobs,'historical_bytes':totalbytes,'build_files':built,'public_anon_jwt_matches_ignored':public,'candidates':found,'limitations':['Encoded/unknown-format credentials may be missed. Matches require manual review; no credentials are tested against online services.']}
dest.write_text(json.dumps(result,indent=2));print(json.dumps({k:v for k,v in result.items() if k!='candidates'},ensure_ascii=False));print('Candidates:',len(found))
