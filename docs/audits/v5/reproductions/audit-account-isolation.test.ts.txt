function deferred<T>() {
    let resolve!: (value: T) => void;
    const promise = new Promise<T>((done) => { resolve = done; });
    return { promise, resolve };
}

describe('STATIC-03 — async cross-account cache (scheduling mocks, no backend claims)', () => {
    it('owner control: a completed A read writes only the A cache', async () => {
        const obsA = observationFor(A);
        setStorageNamespace(A);
        harness.range.mockResolvedValue({ data: [mapToRow(obsA, A)], error: null });
        const result = await getObservations();
        expect(result.observations[0].comment).toBe('PRIVATE COMMENT A');
        expect(harness.eq).toHaveBeenCalledWith('user_id', A);
        expect(localStorage.getItem(cacheKey(A))).toContain('PRIVATE COMMENT A');
        expect(localStorage.getItem(cacheKey(B))).toBeNull();
        expect(globalThis.fetch).not.toHaveBeenCalled();
    });

    it('CONFIRMED: a delayed A response overwrites the B cache and is readable offline as B', async () => {
        const response = deferred<{ data: Record<string, unknown>[]; error: null }>();
        const obsA = observationFor(A);
        const obsB = observationFor(B);
        setStorageNamespace(A);
        harness.range.mockReturnValue(response.promise);
        const aRequest = getObservations();
        await waitFor(() => expect(harness.range).toHaveBeenCalledOnce());
        expect(harness.eq).toHaveBeenCalledWith('user_id', A);

        // Exact scheduling boundary: AuthContext changes this namespace when B signs in.
        setStorageNamespace(B);
        harness.getUser.mockResolvedValue({ data: { user: sessionFor(B).user }, error: null });
        localStorage.setItem(cacheKey(B), JSON.stringify([obsB]));
        response.resolve({ data: [mapToRow(obsA, A)], error: null });
        await aRequest;

        const poisonedCache = JSON.parse(localStorage.getItem(cacheKey(B)) || '[]');
        expect(poisonedCache).toHaveLength(1);
        expect(poisonedCache[0].id).toBe(obsA.id);
        expect(poisonedCache[0].comment).toBe('PRIVATE COMMENT A');
        expect(poisonedCache[0].photo).toBe(PHOTO_A);
        expect(localStorage.getItem(cacheKey(A))).toBeNull();
        Object.defineProperty(navigator, 'onLine', { configurable: true, value: false });
        const asB = await getObservations();
        expect(asB.source).toBe('cache');
        expect(asB.observations[0].gps.lat).toBe(48.12345);
        expect(asB.observations[0].comment).toBe('PRIVATE COMMENT A');
        expect(globalThis.fetch).not.toHaveBeenCalled();
    });
});

describe('AuthContext signOut — locked SDK control versus exceptional throw', () => {
    it('PROTECTED: actual auth-js clears token and AuthContext user even when HTTP logout returns 503', async () => {
        const sdkStorageKey = 'audit-sdk-session';
        localStorage.setItem(sdkStorageKey, JSON.stringify(sessionFor(A)));
        const syntheticTransport = vi.fn(async (input: RequestInfo | URL) => {
            // In-memory response only. Never forwards to fetch or any socket.
            expect(String(input)).toMatch(/^http:\/\/127\.0\.0\.1:1\/auth\/v1\/logout/);
            return new Response(JSON.stringify({ message: 'synthetic unavailable' }), {
                status: 503, headers: { 'Content-Type': 'application/json' },
            });
        });
        const sdk = new GoTrueClient({
            url: 'http://127.0.0.1:1/auth/v1',
            storage: localStorage, storageKey: sdkStorageKey,
            autoRefreshToken: false, persistSession: true, detectSessionInUrl: false,
            fetch: syntheticTransport,
        });
        await sdk.initialize();
        const { data: { subscription } } = sdk.onAuthStateChange((event, session) => {
            if (event === 'SIGNED_OUT') harness.listener?.(event, session);
        });
        harness.signOut.mockImplementation(() => sdk.signOut());
        const { result } = renderHook(() => useAuth(), { wrapper });
        await waitFor(() => expect(result.current.user?.id).toBe(A));
        await act(async () => { await result.current.signOut(); });
        expect(syntheticTransport).toHaveBeenCalledOnce();
        const sdkResult = await harness.signOut.mock.results[0].value;
        expect(sdkResult.error).not.toBeNull();
        expect(result.current.user).toBeNull();
        expect(result.current.session).toBeNull();
        expect(localStorage.getItem(sdkStorageKey)).toBeNull();
        expect(localStorage.getItem('cached_auth_session')).toBeNull();
        expect(globalThis.fetch).not.toHaveBeenCalled();
        subscription.unsubscribe();
        sdk.stopAutoRefresh();
    });

    it('ROBUSTNESS ONLY: an injected exceptional throw removes fallback token but retains React user', async () => {
        // This is NOT the ordinary HTTP/network failure of the locked SDK above.
        // It only documents the application catch path, without a backend exploit claim.
        vi.stubEnv('VITE_SUPABASE_URL', 'https://audit-only.supabase.co');
        const tokenKey = 'sb-audit-only-auth-token';
        localStorage.setItem(tokenKey, 'synthetic-token');
        harness.signOut.mockRejectedValue(new Error('AUDIT_INJECTED_EXCEPTION'));
        const { result } = renderHook(() => useAuth(), { wrapper });
        await waitFor(() => expect(result.current.user?.id).toBe(A));
        await act(async () => { await result.current.signOut(); });
        expect(localStorage.getItem(tokenKey)).toBeNull();
        expect(localStorage.getItem('cached_auth_session')).toBeNull();
        expect(result.current.user?.id).toBe(A);
        expect(result.current.session?.user.id).toBe(A);
        expect(globalThis.fetch).not.toHaveBeenCalled();
    });
});
