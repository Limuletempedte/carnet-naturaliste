const { chromium } = await import(process.env.PLAYWRIGHT_MODULE || 'playwright');
import fs from 'node:fs';
const base='http://127.0.0.1:3000';
const accounts=JSON.parse(fs.readFileSync(new URL('./accounts.json',import.meta.url)));
const browser=await chromium.launch({executablePath:process.env.CHROMIUM_PATH || undefined,headless:true,args:['--no-sandbox']});
const context=await browser.newContext({viewport:{width:1440,height:1000},serviceWorkers:'block'});
const external=[];
await context.route('**/*',async route=>{
 const u=new URL(route.request().url());
 if(['127.0.0.1','localhost'].includes(u.hostname)) return route.continue();
 external.push(u.origin);
 return route.fulfill({status:200,contentType:'application/json',body:JSON.stringify({results:[],features:[],query:{pages:{}},hourly:{time:[],weather_code:[]}})});
});
const page=await context.newPage();
page.setDefaultTimeout(12000);
page.on('dialog',d=>d.accept());
page.on('pageerror',e=>console.log('PAGEERROR',e.message));
page.on('console',m=>{if(m.type()==='error')console.log('CONSOLE_ERROR',m.text());});
await page.goto(base);
async function login(u){
 await page.locator('#login-email').fill(u.email);
 await page.locator('#login-password').fill(u.password);
 await page.locator('form button[type=submit]').click();
 await page.locator('#login-email').waitFor({state:'hidden'});
}
await login(accounts.A);
await page.getByRole('button',{name:"Voir l'observation de PRIVE_A",exact:true}).waitFor();
const understood=page.getByRole('button',{name:'Compris',exact:true});
if(await understood.isVisible()) await understood.click();
await page.getByRole('button',{name:"Voir l'observation de PRIVE_A",exact:true}).click();
await page.getByRole('button',{name:'Modifier',exact:true}).click();
await page.locator('#speciesName').waitFor();
await page.waitForTimeout(300);
console.log('A_BODY', (await page.locator('body').innerText()).slice(0,2500));
console.log('A_FIELDS',await page.locator('input,textarea').evaluateAll(els=>els.map(e=>({name:e.name,id:e.id,value:e.value}))));
await page.getByRole('button',{name:'Menu utilisateur'}).click();
await page.getByRole('menuitem',{name:'Se déconnecter',exact:true}).click();
await page.locator('#login-email').waitFor();
await login(accounts.B);
await page.waitForTimeout(1000);
console.log('B_BODY',(await page.locator('body').innerText()).slice(0,2500));
await page.waitForTimeout(600);
const fields=await page.locator('input,textarea').evaluateAll(els=>els.map(e=>({name:e.name,id:e.id,value:e.value})));
const leaked=fields.some(e=>e.value==='PRIVE_A')&&fields.some(e=>e.value==='SECRET_FICTIF_PRIVE_A');
const result={scenario:'A ouvre Modifier, se déconnecte ; B se connecte dans le même onglet',accountBMenu:(await page.getByRole('button',{name:'Menu utilisateur'}).innerText()).trim(),leaked,fields,externalMocked:Array.from(new Set(external)),serviceWorker:'bloqué pour ce scénario; état React testé sur le bundle réel'};
fs.writeFileSync(new URL('./browser-result.json',import.meta.url),JSON.stringify(result,null,2));
console.log('B_LEAK_CONFIRMED',leaked,fields);
await page.screenshot({path:new URL('./browser-account-leak.png',import.meta.url).pathname,fullPage:true});
console.log('EXTERNAL_MOCKED',Array.from(new Set(external)));
await browser.close();
